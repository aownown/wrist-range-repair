"""Current-arm source/capability/initial-context encoders and trajectory decoder.
This is an open-loop full-source editor scaffold, not proof of local rejoining,
closed-loop recovery, learned improvement, dynamic safety, or novel architecture.
"""
from dataclasses import dataclass
import torch
from torch import nn
from public_arm_editor_representation_v1 import SOURCE_DIM,CONTEXT_DIM,CAPABILITY_DIM


@dataclass(frozen=True)
class Config:
    width:int=64
    heads:int=4
    source_layers:int=2
    decoder_layers:int=2
    feedforward:int=128


def relative_rotation(v):
    if v.shape[-1]!=3:raise ValueError('Axis-angle vector required')
    z=torch.zeros_like(v[...,0]);x,y,w=v.unbind(-1)
    skew=torch.stack((z,-w,y,w,z,-x,-y,x,z),dim=-1).reshape(*v.shape[:-1],3,3)
    return torch.linalg.matrix_exp(skew)


def bounded_reference(latent, initial, bounds):
    lo,hi=bounds[:,None,:,0]+.02,bounds[:,None,:,1]-.02;q0=initial[:,None]
    if latent.shape[-1]!=7 or torch.any(lo>=hi) or torch.any(q0<=lo) or torch.any(q0>=hi):raise ValueError('Original reference strictly inside unchanged 0.02rad guard inset required')
    # Differentiable reference generation, not clipping or moving robot state.
    displacement=torch.tanh(latent)
    return q0+torch.where(displacement>=0,hi-q0,q0-lo)*displacement


class SourceCapabilityEditor(nn.Module):
    def __init__(self,config=Config()):
        super().__init__();self.config=config;d=config.width
        if d%config.heads or min(config.source_layers,config.decoder_layers)<1:raise ValueError('Valid transformer configuration required')
        def mlp(n):return nn.Sequential(nn.Linear(n,d),nn.GELU(),nn.Linear(d,d),nn.LayerNorm(d))
        self.source_projection=nn.Linear(SOURCE_DIM,d)
        layer=nn.TransformerEncoderLayer(d,config.heads,config.feedforward,dropout=0.,activation='gelu',batch_first=True,norm_first=True)
        self.source_encoder=nn.TransformerEncoder(layer,config.source_layers,norm=nn.LayerNorm(d),enable_nested_tensor=False)
        self.capability_encoder=mlp(CAPABILITY_DIM);self.context_encoder=mlp(CONTEXT_DIM)
        self.query_encoder=mlp(4)
        decoder=nn.TransformerDecoderLayer(d,config.heads,config.feedforward,dropout=0.,activation='gelu',batch_first=True,norm_first=True)
        self.edit_decoder=nn.TransformerDecoder(decoder,config.decoder_layers,norm=nn.LayerNorm(d))
        self.orientation_head=nn.Linear(d,3);self.reference_head=nn.Linear(d,7)
        # Exact zero initial edits: source copy is a visible strong starting point.
        for head in (self.orientation_head,self.reference_head):nn.init.zeros_(head.weight);nn.init.zeros_(head.bias)

    def forward(self,batch):
        expected={'source','context','capability','query_time','valid_mask','initial_reference','bounds'}
        if set(batch)!=expected:raise ValueError('Explicit no-label input schema required')
        s=batch['source'];valid=batch['valid_mask'];t=batch['query_time'];b,n,_=s.shape
        if s.shape[-1]!=SOURCE_DIM or valid.dtype!=torch.bool or valid.shape!=(b,n) or t.shape!=(b,n) or torch.any(valid.sum(1)<2):raise ValueError('Original variable-length masked source required')
        for key,value in batch.items():
            if key!='valid_mask' and not torch.isfinite(value).all():raise ValueError('Finite known inputs required')
        # Mask both source encoding AND cross-attention, not only decoder queries.
        memory=self.source_encoder(self.source_projection(s),src_key_padding_mask=~valid)
        cap=self.capability_encoder(batch['capability']);context=self.context_encoder(batch['context'])
        memory=torch.cat((memory,cap[:,None],context[:,None]),dim=1)
        memory_mask=torch.cat((~valid,torch.zeros((b,2),dtype=torch.bool,device=s.device)),dim=1)
        time_features=torch.stack((t,t*t,torch.sin(torch.pi*t),torch.cos(torch.pi*t)),dim=-1)
        query=self.query_encoder(time_features)+cap[:,None]+context[:,None]
        hidden=self.edit_decoder(query,memory,tgt_key_padding_mask=~valid,memory_key_padding_mask=memory_mask)
        v=self.orientation_head(hidden);latent=self.reference_head(hidden);q=bounded_reference(latent,batch['initial_reference'],batch['bounds'])
        return dict(reference_latent=torch.where(valid[...,None],latent,torch.zeros_like(latent)),rotation_delta=torch.where(valid[...,None],v,torch.zeros_like(v)),joint_references=torch.where(valid[...,None],q,batch['initial_reference'][:,None]))


def imitation_loss(prediction,target,valid,bounds,velocity_weight=.05):
    if set(target)!={'rotation_delta','joint_references'} or valid.dtype!=torch.bool or not valid.any():raise ValueError('Separated supervision and nonempty valid mask required')
    v,q=prediction['rotation_delta'],prediction['joint_references']
    if v.shape!=target['rotation_delta'].shape or q.shape!=target['joint_references'].shape or v.shape[:2]!=valid.shape:raise ValueError('Targets must match original action clock')
    rotation_error=relative_rotation(v)-relative_rotation(target['rotation_delta'])
    span=bounds[...,1]-bounds[...,0]
    ref_error=(q-target['joint_references'])/span[:,None]
    rotation=(rotation_error.square()*valid[...,None,None]).sum()/(9*valid.sum())
    reference=(ref_error.square()*valid[...,None]).sum()/(7*valid.sum())
    adjacent=valid[:,1:]&valid[:,:-1]
    difference=(ref_error[:,1:]-ref_error[:,:-1]).square()
    velocity=(difference*adjacent[...,None]).sum()/(7*adjacent.sum().clamp_min(1))
    return rotation+reference+velocity_weight*velocity,dict(rotation=rotation.detach(),reference=reference.detach(),reference_difference=velocity.detach())
